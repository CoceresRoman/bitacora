# bitacora Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a self-hosted NestJS service that turns real technical activity into Spanish LinkedIn posts, gated by human approval over Telegram.

**Architecture:** Modular monolith with ports and adapters. The core domain (entities, state machine, curation, style guard) has zero external dependencies and is tested without network. Four ports — `LlmPort`, `PublisherPort`, `ReviewChannelPort`, `SourcePort` — isolate Claude CLI, LinkedIn, Telegram and the content sources. Persistence is Prisma over SQLite. Scheduling is in-process; there is no Redis and no message broker.

**Tech Stack:** Node.js 22, TypeScript 5.6, NestJS 11, Prisma 6 + SQLite, Jest 29 + ts-jest, Zod 3 for config validation, `node:crypto` for token encryption. No HTTP client library — native `fetch`.

## Global Constraints

- **No Anthropic API.** The only shipped `LlmPort` adapter talks to a local Claude CLI bridge. `ANTHROPIC_API_KEY` must never appear in code, docs, `.env.example` or tests.
- **Posts are written in Spanish.** Repo, code, comments, commit messages and README are in English.
- **The bot must refuse every chat except one.** `TELEGRAM_ALLOWED_CHAT_ID` is required; the process must fail to boot without it.
- **LinkedIn tokens are encrypted at rest** with AES-256-GCM and never logged, not even truncated.
- **The Claude bridge binds to `127.0.0.1` only** and requires the `x-bridge-secret` header.
- **Nothing publishes without an explicit human approval.** A draft with no answer expires after 48 h.
- **Prisma over SQLite does not support enums.** All status/source columns are `String`; the union types live in TypeScript and are validated at the repository boundary.
- **TDD throughout:** failing test first, verified failing, then minimal implementation, then green, then commit. Coverage target 80%.
- Repo: `git@github.com:CoceresRoman/bitacora.git`, branch `main`, MIT license.
- Spec: `docs/superpowers/specs/2026-08-04-bitacora-design.md`.

```ts
import { createHash } from 'node:crypto';

export type SignalSource = 'git' | 'inbox' | 'trends';

export const SIGNAL_SOURCES: readonly SignalSource[] = ['git', 'inbox', 'trends'] as const;

export interface RawSignal {
  id: string;
  source: SignalSource;
  kind: string;
  payload: string;
  refs: string[];
  capturedAt: Date;
  hash: string;
}

export function signalHash(source: SignalSource, payload: string): string {
  return createHash('sha256').update(`${source}:${payload.trim()}`).digest('hex');
}

export function isSignalSource(value: string): value is SignalSource {
  return (SIGNAL_SOURCES as readonly string[]).includes(value);
}
```

```ts
export type DraftStatus =
  | 'DRAFTING'
  | 'PENDING_REVIEW'
  | 'REVISING'
  | 'APPROVED'
  | 'SCHEDULED'
  | 'PUBLISHED'
  | 'DISCARDED'
  | 'EXPIRED';

export type Decision =
  | { kind: 'approve' }
  | { kind: 'discard' }
  | { kind: 'revise'; feedback: string }
  | { kind: 'schedule'; at: Date };

export interface Revision {
  feedback: string;
  body: string;
  at: Date;
}

export interface Draft {
  id: string;
  signalId: string;
  topic: string;
  bodyEs: string;
  status: DraftStatus;
  revisions: Revision[];
  scheduledAt: Date | null;
  publishedUrn: string | null;
  createdAt: Date;
  updatedAt: Date;
}

export const MAX_REVISIONS = 3;
export const REVIEW_EXPIRY_HOURS = 48;

const ALLOWED: Record<DraftStatus, readonly DraftStatus[]> = {
  DRAFTING: ['PENDING_REVIEW', 'DISCARDED'],
  PENDING_REVIEW: ['APPROVED', 'SCHEDULED', 'REVISING', 'DISCARDED', 'EXPIRED'],
  REVISING: ['PENDING_REVIEW', 'DISCARDED'],
  APPROVED: ['PUBLISHED', 'DISCARDED'],
  SCHEDULED: ['PUBLISHED', 'DISCARDED'],
  PUBLISHED: [],
  DISCARDED: [],
  EXPIRED: [],
};

export function canTransition(from: DraftStatus, to: DraftStatus): boolean {
  return ALLOWED[from].includes(to);
}

function transition(draft: Draft, to: DraftStatus, now: Date, patch: Partial<Draft> = {}): Draft {
  if (!canTransition(draft.status, to)) {
    throw new Error(`Invalid transition: ${draft.status} -> ${to}`);
  }
  return { ...draft, ...patch, status: to, updatedAt: now };
}

export function applyDecision(draft: Draft, decision: Decision, now: Date): Draft {
  switch (decision.kind) {
    case 'approve':
      return transition(draft, 'APPROVED', now, { scheduledAt: null });
    case 'schedule':
      return transition(draft, 'SCHEDULED', now, { scheduledAt: decision.at });
    case 'discard':
      return transition(draft, 'DISCARDED', now);
    case 'revise': {
      if (draft.revisions.length >= MAX_REVISIONS) {
        throw new Error(`Max revisions reached (${MAX_REVISIONS})`);
      }
      return transition(draft, 'REVISING', now, {
        revisions: [...draft.revisions, { feedback: decision.feedback, body: draft.bodyEs, at: now }],
      });
    }
  }
}
```

```ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from './prisma.service';
import { isSignalSource, RawSignal } from '../core/signal.entity';

interface SignalRow {
  id: string;
  source: string;
  kind: string;
  payload: string;
  refs: string;
  hash: string;
  capturedAt: Date;
}

function toDomain(row: SignalRow): RawSignal {
  if (!isSignalSource(row.source)) {
    throw new Error(`Unknown signal source in database: ${row.source}`);
  }
  return {
    id: row.id,
    source: row.source,
    kind: row.kind,
    payload: row.payload,
    refs: JSON.parse(row.refs) as string[],
    capturedAt: row.capturedAt,
    hash: row.hash,
  };
}

@Injectable()
export class SignalRepository {
  constructor(private readonly prisma: PrismaService) {}

  async saveNew(signals: RawSignal[]): Promise<RawSignal[]> {
    const inserted: RawSignal[] = [];
    for (const signal of signals) {
      const existing = await this.prisma.signal.findUnique({ where: { hash: signal.hash } });
      if (existing) continue;
      const row = await this.prisma.signal.create({
        data: {
          source: signal.source,
          kind: signal.kind,
          payload: signal.payload,
          refs: JSON.stringify(signal.refs),
          hash: signal.hash,
          capturedAt: signal.capturedAt,
        },
      });
      inserted.push(toDomain(row));
    }
    return inserted;
  }

  async findUnused(): Promise<RawSignal[]> {
    const rows = await this.prisma.signal.findMany({
      where: { usedAt: null },
      orderBy: { capturedAt: 'desc' },
    });
    return rows.map(toDomain);
  }

  async markUsed(id: string): Promise<void> {
    await this.prisma.signal.update({ where: { id }, data: { usedAt: new Date() } });
  }
}
```

```ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from './prisma.service';
import { Draft, DraftStatus, REVIEW_EXPIRY_HOURS, Revision } from '../core/draft.entity';

const STATUSES: readonly DraftStatus[] = [
  'DRAFTING',
  'PENDING_REVIEW',
  'REVISING',
  'APPROVED',
  'SCHEDULED',
  'PUBLISHED',
  'DISCARDED',
  'EXPIRED',
];

interface DraftRow {
  id: string;
  signalId: string;
  topic: string;
  bodyEs: string;
  status: string;
  revisions: string;
  scheduledAt: Date | null;
  publishedUrn: string | null;
  createdAt: Date;
  updatedAt: Date;
}

function toDomain(row: DraftRow): Draft {
  if (!(STATUSES as readonly string[]).includes(row.status)) {
    throw new Error(`Unknown draft status in database: ${row.status}`);
  }
  const revisions = (JSON.parse(row.revisions) as Array<Omit<Revision, 'at'> & { at: string }>).map(
    (revision) => ({ ...revision, at: new Date(revision.at) }),
  );
  return { ...row, status: row.status as DraftStatus, revisions };
}

@Injectable()
export class DraftRepository {
  constructor(private readonly prisma: PrismaService) {}

  async create(input: { signalId: string; topic: string; bodyEs: string }): Promise<Draft> {
    const row = await this.prisma.draft.create({ data: { ...input, status: 'DRAFTING' } });
    return toDomain(row);
  }

  async findById(id: string): Promise<Draft | null> {
    const row = await this.prisma.draft.findUnique({ where: { id } });
    return row ? toDomain(row) : null;
  }

  async save(draft: Draft): Promise<Draft> {
    const row = await this.prisma.draft.update({
      where: { id: draft.id },
      data: {
        topic: draft.topic,
        bodyEs: draft.bodyEs,
        status: draft.status,
        revisions: JSON.stringify(draft.revisions),
        scheduledAt: draft.scheduledAt,
        publishedUrn: draft.publishedUrn,
      },
    });
    return toDomain(row);
  }

  async recentTopics(days: number, now: Date): Promise<string[]> {
    const since = new Date(now.getTime() - days * 24 * 60 * 60 * 1000);
    const rows = await this.prisma.draft.findMany({
      where: { createdAt: { gte: since } },
      select: { topic: true },
    });
    return rows.map((row) => row.topic);
  }

  async findDuePublication(now: Date): Promise<Draft[]> {
    const rows = await this.prisma.draft.findMany({
      where: {
        publishedUrn: null,
        OR: [{ status: 'APPROVED' }, { status: 'SCHEDULED', scheduledAt: { lte: now } }],
      },
    });
    return rows.map(toDomain);
  }

  async findExpirable(now: Date): Promise<Draft[]> {
    const cutoff = new Date(now.getTime() - REVIEW_EXPIRY_HOURS * 60 * 60 * 1000);
    const rows = await this.prisma.draft.findMany({
      where: { status: 'PENDING_REVIEW', updatedAt: { lt: cutoff } },
    });
    return rows.map(toDomain);
  }
}
```

```ts
import { Injectable, Logger } from '@nestjs/common';
import { execFile } from 'node:child_process';
import { basename } from 'node:path';
import { promisify } from 'node:util';

const run = promisify(execFile);
// ASCII unit/record separators: they cannot appear in a commit message.
const FIELD = '\x1f';
const RECORD = '\x1e';

export interface GitCommit {
  repo: string;
  sha: string;
  subject: string;
  body: string;
  at: Date;
}

@Injectable()
export class LocalGitReader {
  private readonly logger = new Logger(LocalGitReader.name);

  async read(repoPaths: string[], since: Date): Promise<GitCommit[]> {
    const results = await Promise.all(repoPaths.map((path) => this.readOne(path, since)));
    return results.flat();
  }

  private async readOne(repoPath: string, since: Date): Promise<GitCommit[]> {
    try {
      const { stdout } = await run(
        'git',
        [
          'log',
          `--since=${since.toISOString()}`,
          `--pretty=format:%H${FIELD}%s${FIELD}%b${FIELD}%aI${RECORD}`,
          '--no-merges',
        ],
        { cwd: repoPath, maxBuffer: 10 * 1024 * 1024 },
      );
      return stdout
        .split(RECORD)
        .map((record) => record.trim())
        .filter((record) => record.length > 0)
        .map((record) => {
          const [sha, subject, body, at] = record.split(FIELD);
          return { repo: basename(repoPath), sha, subject, body: body.trim(), at: new Date(at) };
        });
    } catch (error) {
      this.logger.warn(`Skipping ${repoPath}: ${(error as Error).message}`);
      return [];
    }
  }
}
```

```ts
import { Injectable, Logger } from '@nestjs/common';
import { GitCommit } from './local-git.reader';

interface GithubPushEvent {
  type: string;
  created_at: string;
  repo: { name: string };
  payload: { commits?: Array<{ sha: string; message: string }> };
}

@Injectable()
export class GithubEventsReader {
  private readonly logger = new Logger(GithubEventsReader.name);

  async read(username: string, since: Date, token?: string): Promise<GitCommit[]> {
    const headers: Record<string, string> = {
      Accept: 'application/vnd.github+json',
      'User-Agent': 'bitacora',
    };
    if (token) headers.Authorization = `Bearer ${token}`;

    try {
      const response = await fetch(
        `https://api.github.com/users/${encodeURIComponent(username)}/events/public?per_page=100`,
        { headers },
      );
      if (!response.ok) {
        this.logger.warn(`GitHub events request failed with status ${response.status}`);
        return [];
      }
      const events = (await response.json()) as GithubPushEvent[];
      return events
        .filter((event) => event.type === 'PushEvent' && new Date(event.created_at) >= since)
        .flatMap((event) =>
          (event.payload.commits ?? []).map((commit) => ({
            repo: event.repo.name,
            sha: commit.sha,
            subject: commit.message.split('\n')[0],
            body: commit.message.split('\n').slice(1).join('\n').trim(),
            at: new Date(event.created_at),
          })),
        );
    } catch (error) {
      this.logger.warn(`GitHub events unreachable: ${(error as Error).message}`);
      return [];
    }
```

```ts
import { Inject, Injectable } from '@nestjs/common';
import { RawSignal, signalHash, SignalSource } from '../../core/signal.entity';
import { SourcePort } from '../../core/ports/source.port';
import { GitCommit, LocalGitReader } from './local-git.reader';
import { GithubEventsReader } from './github-events.reader';

export interface GitActivityConfig {
  username: string;
  repoPaths: string[];
  token?: string;
}

export const GIT_ACTIVITY_CONFIG = Symbol('GitActivityConfig');

const NOISE = /^(chore|ci|build|style)(\(|:)|^Merge |^Revert /i;

@Injectable()
export class GitActivitySource implements SourcePort {
  readonly source: SignalSource = 'git';

  constructor(
    private readonly local: LocalGitReader,
    private readonly remote: GithubEventsReader,
    @Inject(GIT_ACTIVITY_CONFIG) private readonly config: GitActivityConfig,
  ) {}

  async collect(since: Date): Promise<RawSignal[]> {
    const [local, remote] = await Promise.all([
      this.local.read(this.config.repoPaths, since),
      this.remote.read(this.config.username, since, this.config.token),
    ]);

    const bySha = new Map<string, GitCommit>();
    for (const commit of [...local, ...remote]) {
      if (NOISE.test(commit.subject)) continue;
      if (!bySha.has(commit.sha)) bySha.set(commit.sha, commit);
    }

    return [...bySha.values()].map((commit) => {
      const payload = [commit.subject, commit.body].filter(Boolean).join('\n\n');
      return {
        id: '',
        source: 'git' as const,
        kind: 'commit',
        payload,
        refs: [`${commit.repo}@${commit.sha}`],
        capturedAt: commit.at,
        hash: signalHash('git', payload),
      };
    });
  }
}
```

```ts
import { Inject, Injectable, Logger } from '@nestjs/common';
import { RawSignal, signalHash, SignalSource } from '../../core/signal.entity';
import { SourcePort } from '../../core/ports/source.port';

export interface TrendsConfig {
  repos: string[];
}

export const TRENDS_CONFIG = Symbol('TrendsConfig');

const MIN_HN_POINTS = 150;
const HN_URL = 'https://hn.algolia.com/api/v1/search?tags=front_page';

interface HnHit {
  objectID: string;
  title: string;
  url: string | null;
  points: number;
}

interface GithubRelease {
  tag_name: string;
  name: string | null;
  body: string | null;
  html_url: string;
  published_at: string;
}

@Injectable()
export class TrendsSource implements SourcePort {
  readonly source: SignalSource = 'trends';
  private readonly logger = new Logger(TrendsSource.name);

  constructor(@Inject(TRENDS_CONFIG) private readonly config: TrendsConfig) {}

  async collect(since: Date): Promise<RawSignal[]> {
    const [stories, releases] = await Promise.all([this.hackerNews(), this.releases(since)]);
    return [...stories, ...releases];
  }

  private async hackerNews(): Promise<RawSignal[]> {
    try {
      const response = await fetch(HN_URL);
      if (!response.ok) return [];
      const { hits } = (await response.json()) as { hits: HnHit[] };
      return hits
        .filter((hit) => hit.points >= MIN_HN_POINTS && hit.url)
        .map((hit) => {
          const payload = `${hit.title} (${hit.points} puntos en Hacker News)`;
          return {
            id: '',
            source: 'trends' as const,
            kind: 'hn-story',
            payload,
            refs: [hit.url as string],
            capturedAt: new Date(),
            hash: signalHash('trends', payload),
          };
        });
    } catch (error) {
      this.logger.warn(`Hacker News unreachable: ${(error as Error).message}`);
      return [];
    }
  }

  private async releases(since: Date): Promise<RawSignal[]> {
    const results = await Promise.all(
      this.config.repos.map(async (repo) => {
        try {
          const response = await fetch(`https://api.github.com/repos/${repo}/releases/latest`, {
            headers: { Accept: 'application/vnd.github+json', 'User-Agent': 'bitacora' },
          });
          if (!response.ok) return null;
          const release = (await response.json()) as GithubRelease;
          if (new Date(release.published_at) < since) return null;
          const payload = [
            `${repo} publicó ${release.name ?? release.tag_name}`,
            (release.body ?? '').slice(0, 1000),
          ]
            .filter(Boolean)
            .join('\n\n');
          return {
            id: '',
            source: 'trends' as const,
            kind: 'release',
            payload,
            refs: [release.html_url],
            capturedAt: new Date(release.published_at),
            hash: signalHash('trends', payload),
          };
        } catch (error) {
          this.logger.warn(`Release lookup failed for ${repo}: ${(error as Error).message}`);
          return null;
        }
      }),
    );
    return results.filter((signal): signal is RawSignal => signal !== null);
  }
}
```

```ts
import { Injectable } from '@nestjs/common';
import { RawSignal, signalHash, SignalSource } from '../../core/signal.entity';
import { SourcePort } from '../../core/ports/source.port';
import { SignalRepository } from '../../persistence/signal.repository';

@Injectable()
export class InboxSource implements SourcePort {
  readonly source: SignalSource = 'inbox';

  constructor(private readonly signals: SignalRepository) {}

  async capture(text: string, now: Date): Promise<void> {
    const payload = text.trim();
    if (payload.length === 0) {
      throw new Error('Cannot capture an empty idea');
    }
    await this.signals.saveNew([
      {
        id: '',
        source: 'inbox',
        kind: 'idea',
        payload,
        refs: [],
        capturedAt: now,
        hash: signalHash('inbox', payload),
      },
    ]);
  }

  async collect(_since: Date): Promise<RawSignal[]> {
    const unused = await this.signals.findUnused();
    return unused.filter((signal) => signal.source === 'inbox');
  }
}
```

**Scoring rules** — deterministic and unit-tested, no LLM involved:
- Base by source: `inbox` 100 (you asked for it explicitly), `git` 60, `trends` 30.
- Recency bonus: `+20` if captured within 48 h, `+10` within 7 days, `0` beyond.
- Substance bonus: `+15` when the payload is longer than 120 characters.
- Topic-overlap penalty: `-1000` when a significant word of the payload matches any of the recent topics. That is effectively a veto, not a nudge.
- A candidate scoring below `0` is never returned. If every candidate is vetoed, `pickBest` returns `null` and the cycle skips.

```ts
import { RawSignal, SignalSource } from '../core/signal.entity';
import { SOURCE_PORTS, SourcePort } from '../core/ports/source.port';
import { SignalRepository } from '../persistence/signal.repository';
import { DraftRepository } from '../persistence/draft.repository';

export const TOPIC_COOLDOWN_DAYS = 14;

const BASE_SCORE: Record<SignalSource, number> = { inbox: 100, git: 60, trends: 30 };
const SUBSTANCE_MIN_LENGTH = 120;
const TOPIC_VETO = -1000;
const SIGNIFICANT_WORD_MIN_LENGTH = 5;
const DAY_MS = 24 * 60 * 60 * 1000;

function significantWords(text: string): string[] {
  return text
    .toLowerCase()
    .split(/[^a-záéíóúñ0-9]+/i)
    .filter((word) => word.length >= SIGNIFICANT_WORD_MIN_LENGTH);
}

export function scoreSignal(signal: RawSignal, recentTopics: string[], now: Date): number {
  const words = new Set(significantWords(signal.payload));
  const collides = recentTopics
    .flatMap((topic) => significantWords(topic))
    .some((word) => words.has(word));
  if (collides) return TOPIC_VETO;

  const ageMs = now.getTime() - signal.capturedAt.getTime();
  const recency = ageMs <= 2 * DAY_MS ? 20 : ageMs <= 7 * DAY_MS ? 10 : 0;
  const substance = signal.payload.length > SUBSTANCE_MIN_LENGTH ? 15 : 0;

  return BASE_SCORE[signal.source] + recency + substance;
}

@Injectable()
export class CuratorService {
  private readonly logger = new Logger(CuratorService.name);

  constructor(
    @Inject(SOURCE_PORTS) private readonly sources: SourcePort[],
    private readonly signals: SignalRepository,
    private readonly drafts: DraftRepository,
  ) {}

  async collect(since: Date): Promise<RawSignal[]> {
    const batches = await Promise.all(
      this.sources.map(async (source) => {
        try {
          return await source.collect(since);
        } catch (error) {
          this.logger.warn(`Source ${source.source} failed: ${(error as Error).message}`);
          return [];
        }
      }),
    );
    return this.signals.saveNew(batches.flat());
  }

  pickBest(candidates: RawSignal[], recentTopics: string[], now: Date): RawSignal | null {
    const ranked = candidates
      .map((signal) => ({ signal, score: scoreSignal(signal, recentTopics, now) }))
      .filter((entry) => entry.score > 0)
      .sort((a, b) => b.score - a.score);
    return ranked[0]?.signal ?? null;
  }

  async recentTopics(now: Date): Promise<string[]> {
    return this.drafts.recentTopics(TOPIC_COOLDOWN_DAYS, now);
  }
}
```

**Bridge contract** — this is the only place the Claude CLI is invoked:
- `POST /generate`, header `x-bridge-secret`, body `{ "prompt": string }`
- `200 { "text": string }` on success, `401` on a bad secret, `500 { "error": string }` on CLI failure
- Binds to `127.0.0.1` only. Never to `0.0.0.0`.
- Runs `claude -p --output-format json` with the prompt on stdin and reads the `result` field of the JSON response.

```js
#!/usr/bin/env node
// Local bridge to the Claude CLI. Uses your Claude subscription — no API key.
// Start with: CLAUDE_BRIDGE_SECRET=... npm run bridge
import { createServer } from 'node:http';
import { spawn } from 'node:child_process';

const PORT = Number(process.env.CLAUDE_BRIDGE_PORT ?? 8788);
const SECRET = process.env.CLAUDE_BRIDGE_SECRET;
const TIMEOUT_MS = Number(process.env.CLAUDE_BRIDGE_TIMEOUT_MS ?? 180000);

if (!SECRET) {
  console.error('CLAUDE_BRIDGE_SECRET is required');
  process.exit(1);
}

function runClaude(prompt) {
  return new Promise((resolve, reject) => {
    const child = spawn('claude', ['-p', '--output-format', 'json'], {
      stdio: ['pipe', 'pipe', 'pipe'],
    });
    let stdout = '';
    let stderr = '';
    const timer = setTimeout(() => {
      child.kill('SIGKILL');
      reject(new Error(`claude timed out after ${TIMEOUT_MS}ms`));
    }, TIMEOUT_MS);

    child.stdout.on('data', (chunk) => (stdout += chunk));
    child.stderr.on('data', (chunk) => (stderr += chunk));
    child.on('error', (error) => {
      clearTimeout(timer);
      reject(error);
    });
    child.on('close', (code) => {
      clearTimeout(timer);
      if (code !== 0) return reject(new Error(stderr.trim() || `claude exited with ${code}`));
      try {
        const parsed = JSON.parse(stdout);
        if (typeof parsed.result !== 'string') {
          return reject(new Error('claude response has no string "result" field'));
        }
        resolve(parsed.result);
      } catch (error) {
        reject(new Error(`unparseable claude output: ${error.message}`));
      }
    });

    child.stdin.end(prompt);
  });
}

const server = createServer((req, res) => {
  const send = (status, body) => {
    res.writeHead(status, { 'content-type': 'application/json' });
    res.end(JSON.stringify(body));
  };

  if (req.method !== 'POST' || req.url !== '/generate') return send(404, { error: 'not found' });
  if (req.headers['x-bridge-secret'] !== SECRET) return send(401, { error: 'unauthorized' });

  let raw = '';
  req.on('data', (chunk) => (raw += chunk));
  req.on('end', async () => {
    try {
      const { prompt } = JSON.parse(raw);
      if (typeof prompt !== 'string' || prompt.trim() === '') {
        return send(400, { error: 'prompt is required' });
      }
      send(200, { text: await runClaude(prompt) });
    } catch (error) {
      send(500, { error: error.message });
    }
  });
});

// 127.0.0.1 only. Never bind this to a public interface.
server.listen(PORT, '127.0.0.1', () => {
  console.log(`claude bridge listening on http://127.0.0.1:${PORT}`);
});
```

```ts
import { Inject, Injectable } from '@nestjs/common';
import { LlmPort } from '../core/ports/llm.port';

export interface ClaudeBridgeConfig {
  url: string;
  secret: string;
  timeoutMs: number;
}

export const CLAUDE_BRIDGE_CONFIG = Symbol('ClaudeBridgeConfig');

@Injectable()
export class ClaudeCliAdapter implements LlmPort {
  constructor(@Inject(CLAUDE_BRIDGE_CONFIG) private readonly config: ClaudeBridgeConfig) {}

  async generate(prompt: string): Promise<string> {
    let response: Response;
    try {
      response = await fetch(`${this.config.url}/generate`, {
        method: 'POST',
        headers: {
          'content-type': 'application/json',
          'x-bridge-secret': this.config.secret,
        },
        body: JSON.stringify({ prompt }),
        signal: AbortSignal.timeout(this.config.timeoutMs),
      });
    } catch (error) {
      throw new Error(`Claude bridge unreachable: ${(error as Error).message}`);
    }

    if (!response.ok) {
      throw new Error(`Claude bridge returned ${response.status}: ${await response.text()}`);
    }

    const body = (await response.json()) as { text?: unknown };
    if (typeof body.text !== 'string') {
      throw new Error('Malformed bridge response: missing "text"');
    }
    return body.text;
  }
}
```

```ts
export interface StyleViolation {
  rule: string;
  detail: string;
}

const BANNED_OPENINGS = [
  'en el mundo del',
  'en el mundo de la',
  'emocionado de',
  'feliz de anunciar',
  'hoy quiero compartir',
];

const ENGAGEMENT_BAIT = ['que opinas', 'dejame saber', 'comenta abajo'];

const MAX_HASHTAGS = 3;
const MAX_EMOJI_PER_100_WORDS = 1;
const MIN_LENGTH = 400;
const MAX_LENGTH = 2600;

const EMOJI = /\p{Extended_Pictographic}/gu;

function normalize(text: string): string {
  return text
    .toLowerCase()
    .normalize('NFD')
    .replace(/\p{Diacritic}/gu, '');
}

export function checkStyle(body: string, refs: string[]): StyleViolation[] {
  const violations: StyleViolation[] = [];
  const normalized = normalize(body);
  const trimmed = normalized.trimStart();

  const opening = BANNED_OPENINGS.find((phrase) => trimmed.startsWith(phrase));
  if (opening) {
    violations.push({ rule: 'template-opening', detail: `Empieza con "${opening}"` });
  }

  const hashtags = body.match(/#[\p{L}\d_]+/gu) ?? [];
  if (hashtags.length > MAX_HASHTAGS) {
    violations.push({ rule: 'hashtag-count', detail: `${hashtags.length} hashtags` });
  }

  const words = body.split(/\s+/).filter(Boolean).length;
  const emoji = body.match(EMOJI)?.length ?? 0;
  if (words > 0 && emoji > (words / 100) * MAX_EMOJI_PER_100_WORDS) {
    violations.push({ rule: 'emoji-density', detail: `${emoji} emoji en ${words} palabras` });
  }

  const bait = ENGAGEMENT_BAIT.find((phrase) => normalized.includes(phrase));
  if (bait) {
    violations.push({ rule: 'engagement-bait', detail: `Contiene "${bait}"` });
  }

  const hasRef = refs.some((ref) => body.includes(ref));
  const hasAnchor = /`[^`]+`/.test(body) || /\d/.test(body);
  if (!hasRef && !hasAnchor) {
    violations.push({ rule: 'missing-reference', detail: 'Sin referencia al hecho de origen' });
  }

  if (body.length < MIN_LENGTH || body.length > MAX_LENGTH) {
    violations.push({ rule: 'length', detail: `${body.length} caracteres` });
  }

  return violations;
}
```

```ts
import { Inject, Injectable, Logger } from '@nestjs/common';
import { LLM_PORT, LlmPort } from '../core/ports/llm.port';
import { RawSignal } from '../core/signal.entity';
import { Draft } from '../core/draft.entity';
import { DraftRepository } from '../persistence/draft.repository';
import { buildPrompt } from './prompt.builder';
import { checkStyle, StyleViolation } from './style.guard';

export const VOICE_PROFILE = Symbol('VoiceProfile');

const TOPIC_MAX_LENGTH = 120;

@Injectable()
export class GeneratorService {
  private readonly logger = new Logger(GeneratorService.name);

  /** Violations of the body that was finally stored. Empty when the guard passed. */
  lastViolations: StyleViolation[] = [];

  constructor(
    @Inject(LLM_PORT) private readonly llm: LlmPort,
    private readonly drafts: DraftRepository,
    @Inject(VOICE_PROFILE) private readonly voiceProfile: string,
  ) {}

  async generateFor(signal: RawSignal): Promise<Draft> {
    const first = (await this.llm.generate(buildPrompt({ signal, voiceProfile: this.voiceProfile })))
      .trim();
    let body = first;
    let violations = checkStyle(body, signal.refs);

    if (violations.length > 0) {
      this.logger.warn(`Style guard failed, retrying once: ${violations.map((v) => v.rule).join(', ')}`);
      body = (
        await this.llm.generate(
          buildPrompt({ signal, voiceProfile: this.voiceProfile, previousBody: first, violations }),
        )
      ).trim();
      violations = checkStyle(body, signal.refs);
    }

    this.lastViolations = violations;

    const created = await this.drafts.create({
      signalId: signal.id,
      topic: signal.payload.split('\n')[0].slice(0, TOPIC_MAX_LENGTH),
      bodyEs: body,
    });
    return this.drafts.save({ ...created, bodyEs: body, status: 'PENDING_REVIEW' });
  }

  async regenerate(draft: Draft, feedback: string): Promise<Draft> {
    const signal: RawSignal = {
      id: draft.signalId,
      source: 'git',
      kind: 'revision',
      payload: draft.topic,
      refs: [],
      capturedAt: draft.createdAt,
      hash: '',
    };
    const body = (
      await this.llm.generate(
        buildPrompt({
          signal,
          voiceProfile: this.voiceProfile,
          previousBody: draft.bodyEs,
          feedback,
        }),
      )
    ).trim();

    this.lastViolations = checkStyle(body, []);
    return this.drafts.save({ ...draft, bodyEs: body, status: 'PENDING_REVIEW' });
  }
}
```

**Security requirement — this is the task where it lands.** Every update is checked against `allowedChatId` *before* anything else happens. A message from any other chat is dropped with a log line and no reply — no error message either, since replying confirms the bot exists.

**Interaction design:**
- A draft arrives as a message: the body, then a footer with the source ref and any style violations, then four inline buttons.
- `ok` → `{ kind: 'approve' }`. `del` → `{ kind: 'discard' }`. `sched` → `{ kind: 'schedule', at: next 09:00 local }`.
- `rev` → the adapter stores "awaiting feedback for draft X" and replies asking for the change; the *next* plain text message from the allowed chat becomes `{ kind: 'revise', feedback }`.
- A plain text message with no pending revision is an inbox idea: it goes to `InboxSource.capture()` and gets a confirmation reply.

```ts
  constructor(private readonly botToken: string) {}

  private async call<T>(method: string, body: unknown): Promise<T | null> {
    try {
      const response = await fetch(`https://api.telegram.org/bot${this.botToken}/${method}`, {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify(body),
      });
      if (!response.ok) {
        this.logger.warn(`Telegram ${method} failed with status ${response.status}`);
        return null;
      }
      const payload = (await response.json()) as { ok: boolean; result: T };
      return payload.ok ? payload.result : null;
    } catch (error) {
      this.logger.warn(`Telegram ${method} unreachable: ${(error as Error).message}`);
      return null;
    }
  }

  async getUpdates(offset: number, timeoutSec: number): Promise<TelegramUpdate[]> {
    return (await this.call<TelegramUpdate[]>('getUpdates', { offset, timeout: timeoutSec })) ?? [];

    await this.call('sendMessage', {

      text,
      reply_markup: keyboard,
      disable_web_page_preview: true,
    });
```

```ts
import { Inject, Injectable, Logger, OnModuleInit, OnModuleDestroy } from '@nestjs/common';
import { Decision, Draft } from '../../core/draft.entity';
import { DecisionHandler, ReviewChannelPort } from '../../core/ports/review-channel.port';
import { InboxSource } from '../../sources/inbox/inbox.source';
import { InlineKeyboard, TelegramClient, TelegramUpdate } from './telegram.client';

export interface TelegramConfig {
  botToken: string;
  allowedChatId: string;
}

export const TELEGRAM_CONFIG = Symbol('TelegramConfig');

const POLL_TIMEOUT_SEC = 30;
const SCHEDULE_HOUR = 9;

function nextNineAm(now: Date): Date {
  const next = new Date(now);
  next.setHours(SCHEDULE_HOUR, 0, 0, 0);
  if (next <= now) next.setDate(next.getDate() + 1);
  return next;
}

@Injectable()
export class TelegramAdapter implements ReviewChannelPort, OnModuleInit, OnModuleDestroy {
  private readonly logger = new Logger(TelegramAdapter.name);
  private handler: DecisionHandler | null = null;
  private awaitingFeedbackFor: string | null = null;
  private offset = 0;
  private polling = false;

  constructor(
    private readonly client: TelegramClient,
    private readonly inbox: InboxSource,
    @Inject(TELEGRAM_CONFIG) private readonly config: TelegramConfig,
  ) {}

  onModuleInit(): void {
    this.polling = true;
    void this.pollLoop();
  }

  onModuleDestroy(): void {
    this.polling = false;
  }

  private async pollLoop(): Promise<void> {
    while (this.polling) {
      const updates = await this.client.getUpdates(this.offset, POLL_TIMEOUT_SEC);
      for (const update of updates) {
        this.offset = update.update_id + 1;
        try {
          await this.handleUpdate(update, new Date());
        } catch (error) {
          this.logger.error(`Update ${update.update_id} failed: ${(error as Error).message}`);
        }
      }
    }
  }

  onDecision(handler: DecisionHandler): void {
    this.handler = handler;
  }

  async notify(text: string): Promise<void> {
    await this.client.sendMessage(this.config.allowedChatId, text);
  }

  async sendForReview(draft: Draft): Promise<void> {
    const keyboard: InlineKeyboard = {
      inline_keyboard: [
        [
          { text: '✅ Aprobar', callback_data: `ok:${draft.id}` },
          { text: '✏️ Reescribir', callback_data: `rev:${draft.id}` },
        ],
        [
          { text: '🕘 Programar 09:00', callback_data: `sched:${draft.id}` },
          { text: '🗑 Descartar', callback_data: `del:${draft.id}` },
        ],
      ],
    };
    const footer = `\n\n— tema: ${draft.topic} · ${draft.bodyEs.length} caracteres`;
    await this.client.sendMessage(
      this.config.allowedChatId,
      `${draft.bodyEs}${footer}`,
      keyboard,
    );
  }

  /** Exposed for tests and for the poll loop. Enforces the chat whitelist. */
  async handleUpdate(update: TelegramUpdate, now: Date): Promise<void> {
    const chatId =
      update.callback_query?.message?.chat.id ?? update.message?.chat.id ?? undefined;
    if (chatId === undefined || String(chatId) !== this.config.allowedChatId) {
      this.logger.warn(`Dropping update from unauthorized chat ${String(chatId)}`);
      return;
    }

    if (update.callback_query?.data) {
      await this.handleCallback(update.callback_query.id, update.callback_query.data, now);
      return;
    }

    const text = update.message?.text?.trim();
    if (!text) return;

    if (this.awaitingFeedbackFor) {
      const draftId = this.awaitingFeedbackFor;
      this.awaitingFeedbackFor = null;
      await this.emit(draftId, { kind: 'revise', feedback: text });
      return;
    }

    await this.inbox.capture(text, now);
    await this.notify('Idea guardada. Entra al próximo ciclo.');
  }

  private async handleCallback(callbackId: string, data: string, now: Date): Promise<void> {
    const [action, draftId] = data.split(':');
    switch (action) {
      case 'ok':
        await this.client.answerCallbackQuery(callbackId, 'Aprobado');
        await this.emit(draftId, { kind: 'approve' });
        return;
      case 'del':
        await this.client.answerCallbackQuery(callbackId, 'Descartado');
        await this.emit(draftId, { kind: 'discard' });
        return;
      case 'sched':
        await this.client.answerCallbackQuery(callbackId, 'Programado 09:00');
        await this.emit(draftId, { kind: 'schedule', at: nextNineAm(now) });
        return;
      case 'rev':
        this.awaitingFeedbackFor = draftId;
        await this.client.answerCallbackQuery(callbackId, 'Mandá el cambio');
```

```ts
import { createCipheriv, createDecipheriv, randomBytes } from 'node:crypto';

const ALGORITHM = 'aes-256-gcm';
const IV_BYTES = 12;
const KEY_BYTES = 32;

function toKey(keyHex: string): Buffer {
  const key = Buffer.from(keyHex, 'hex');
  if (key.length !== KEY_BYTES) {
    throw new Error(`ENCRYPTION_KEY must decode to 32 bytes, got ${key.length}`);
  }
  return key;
}

export function encrypt(plaintext: string, keyHex: string): string {
  const iv = randomBytes(IV_BYTES);
  const cipher = createCipheriv(ALGORITHM, toKey(keyHex), iv);
  const encrypted = Buffer.concat([cipher.update(plaintext, 'utf8'), cipher.final()]);
  return [iv.toString('hex'), cipher.getAuthTag().toString('hex'), encrypted.toString('hex')].join(
    ':',
  );
}

export function decrypt(payload: string, keyHex: string): string {
  const [ivHex, tagHex, cipherHex] = payload.split(':');
  if (!ivHex || !tagHex || !cipherHex) {
    throw new Error('Malformed encrypted payload');
  }
  const decipher = createDecipheriv(ALGORITHM, toKey(keyHex), Buffer.from(ivHex, 'hex'));
  decipher.setAuthTag(Buffer.from(tagHex, 'hex'));
  return Buffer.concat([
    decipher.update(Buffer.from(cipherHex, 'hex')),
    decipher.final(),
  ]).toString('utf8');
}
```

```ts
import { Inject, Injectable } from '@nestjs/common';
import { PrismaService } from '../../persistence/prisma.service';
import { decrypt, encrypt } from '../crypto';

export const ENCRYPTION_KEY = Symbol('EncryptionKey');
export const REAUTH_WARNING_DAYS = 5;

const SINGLETON_ID = 'singleton';
const DAY_MS = 24 * 60 * 60 * 1000;

export interface LinkedInToken {
  accessToken: string;
  memberUrn: string;
  expiresAt: Date;
}

@Injectable()
export class TokenStore {
  constructor(
    private readonly prisma: PrismaService,
    @Inject(ENCRYPTION_KEY) private readonly keyHex: string,
  ) {}

  async save(token: LinkedInToken): Promise<void> {
    const data = {
      encryptedAccessToken: encrypt(token.accessToken, this.keyHex),
      memberUrn: token.memberUrn,
      expiresAt: token.expiresAt,
    };
    await this.prisma.linkedInToken.upsert({
      where: { id: SINGLETON_ID },
      create: { id: SINGLETON_ID, ...data },
      update: data,
    });
  }

  async current(): Promise<LinkedInToken | null> {
    const row = await this.prisma.linkedInToken.findUnique({ where: { id: SINGLETON_ID } });
    if (!row) return null;
    return {
      accessToken: decrypt(row.encryptedAccessToken, this.keyHex),
      memberUrn: row.memberUrn,
      expiresAt: row.expiresAt,
    };
  }

  async needsReauth(now: Date): Promise<boolean> {
    const row = await this.prisma.linkedInToken.findUnique({ where: { id: SINGLETON_ID } });
    if (!row) return true;
    return row.expiresAt.getTime() - now.getTime() <= REAUTH_WARNING_DAYS * DAY_MS;
  }
}
```

```ts
import { Controller, Get, Query, Res, Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { randomBytes } from 'node:crypto';
import type { Response } from 'express';
import type { Env } from '../../config/env.schema';
import { TokenStore } from './token.store';

const AUTH_URL = 'https://www.linkedin.com/oauth/v2/authorization';
const TOKEN_URL = 'https://www.linkedin.com/oauth/v2/accessToken';
const USERINFO_URL = 'https://api.linkedin.com/v2/userinfo';
const SCOPES = 'openid profile w_member_social';

@Controller('auth/linkedin')
export class LinkedInOAuthController {
  private readonly logger = new Logger(LinkedInOAuthController.name);
  private pendingState: string | null = null;

  constructor(
    private readonly config: ConfigService<Env, true>,
    private readonly tokens: TokenStore,
  ) {}

  @Get()
  start(@Res() res: Response): void {
    this.pendingState = randomBytes(16).toString('hex');
    const params = new URLSearchParams({
      response_type: 'code',
      client_id: this.config.get('LINKEDIN_CLIENT_ID', { infer: true }),
      redirect_uri: this.config.get('LINKEDIN_REDIRECT_URI', { infer: true }),
      state: this.pendingState,
      scope: SCOPES,
    });
    res.redirect(`${AUTH_URL}?${params.toString()}`);
  }

  @Get('callback')
  async callback(@Query('code') code: string, @Query('state') state: string): Promise<string> {
    if (!this.pendingState || state !== this.pendingState) {
      return 'Invalid state. Start again at /auth/linkedin';
    }
    this.pendingState = null;

    const tokenResponse = await fetch(TOKEN_URL, {
      method: 'POST',
      headers: { 'content-type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams({
        grant_type: 'authorization_code',
        code,
        redirect_uri: this.config.get('LINKEDIN_REDIRECT_URI', { infer: true }),
        client_id: this.config.get('LINKEDIN_CLIENT_ID', { infer: true }),
        client_secret: this.config.get('LINKEDIN_CLIENT_SECRET', { infer: true }),
      }),
    });
    if (!tokenResponse.ok) {
      this.logger.error(`Token exchange failed with status ${tokenResponse.status}`);
      return 'Token exchange failed. Check the server logs.';
    }
    const token = (await tokenResponse.json()) as { access_token: string; expires_in: number };

    const userinfo = await fetch(USERINFO_URL, {
      headers: { Authorization: `Bearer ${token.access_token}` },
    });
    if (!userinfo.ok) {
      this.logger.error(`Userinfo failed with status ${userinfo.status}`);
      return 'Could not read the member id. Check the server logs.';
    }
    const { sub } = (await userinfo.json()) as { sub: string };

    await this.tokens.save({
      accessToken: token.access_token,
      memberUrn: `urn:li:person:${sub}`,
      expiresAt: new Date(Date.now() + token.expires_in * 1000),
    });

    return 'LinkedIn connected. You can close this tab.';
  }
}
```

```ts
import { Inject, Injectable } from '@nestjs/common';
import { PublishedRef, PublisherPort } from '../../core/ports/publisher.port';
import { TokenStore } from './token.store';

export const LINKEDIN_API_VERSION = Symbol('LinkedInApiVersion');

const POSTS_URL = 'https://api.linkedin.com/rest/posts';

@Injectable()
export class LinkedInPublisher implements PublisherPort {
  constructor(
    private readonly tokens: TokenStore,
    @Inject(LINKEDIN_API_VERSION) private readonly apiVersion: string,
  ) {}

  async publish(text: string): Promise<PublishedRef> {
    const token = await this.tokens.current();
    if (!token) {
      throw new Error('LinkedIn is not connected. Visit /auth/linkedin first.');
    }

    const response = await fetch(POSTS_URL, {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${token.accessToken}`,
        'content-type': 'application/json',
        'LinkedIn-Version': this.apiVersion,
        'X-Restli-Protocol-Version': '2.0.0',
      },
      body: JSON.stringify({
        author: token.memberUrn,
        commentary: text,
        visibility: 'PUBLIC',
        lifecycleState: 'PUBLISHED',
        distribution: {
          feedDistribution: 'MAIN_FEED',
          targetEntities: [],
          thirdPartyDistributionChannels: [],
        },
        isReshareDisabledByAuthor: false,
      }),
    });

    if (!response.ok) {
      // Never include the token in the message.
      throw new Error(`LinkedIn returned ${response.status}: ${await response.text()}`);
    }

    const urn = response.headers.get('x-restli-id');
    if (!urn) {
      throw new Error('LinkedIn accepted the post but returned no x-restli-id header');
    }
    return { urn };
```

```ts
import { Inject, Injectable, Logger } from '@nestjs/common';
import { Draft } from '../core/draft.entity';
import { PUBLISHER_PORT, PublisherPort } from '../core/ports/publisher.port';
import { REVIEW_CHANNEL_PORT, ReviewChannelPort } from '../core/ports/review-channel.port';
import { DraftRepository } from '../persistence/draft.repository';

export const BACKOFF_MINUTES = [1, 5, 25];
export const MAX_PUBLISH_ATTEMPTS = 3;

@Injectable()
export class PublishQueue {
  private readonly logger = new Logger(PublishQueue.name);

  constructor(
    private readonly drafts: DraftRepository,
    @Inject(PUBLISHER_PORT) private readonly publisher: PublisherPort,
    @Inject(REVIEW_CHANNEL_PORT) private readonly review: ReviewChannelPort,
  ) {}

  async drain(now: Date): Promise<void> {
    const due = await this.drafts.findDuePublication(now);
    for (const draft of due) {
      if (draft.publishedUrn) continue;
      await this.publishOne(draft, now);
    }
  }

  private async publishOne(draft: Draft, now: Date): Promise<void> {
    try {
      const { urn } = await this.publisher.publish(draft.bodyEs);
      await this.drafts.save({ ...draft, status: 'PUBLISHED', publishedUrn: urn, updatedAt: now });
      this.logger.log(`Published draft ${draft.id} as ${urn}`);
    } catch (error) {
      const message = (error as Error).message;
      const attempts = await this.drafts.incrementAttempts(draft.id);
      this.logger.warn(`Publish attempt ${attempts} for ${draft.id} failed: ${message}`);

      if (attempts >= MAX_PUBLISH_ATTEMPTS) {
        await this.review.notify(
          `No pude publicar el draft ${draft.id} después de ${attempts} intentos.\nÚltimo error: ${message}`,
        );
      }
    }
  }

  /** Minutes to wait before the next attempt. Consumed by the scheduler. */
  backoffMinutes(attempts: number): number {
    return BACKOFF_MINUTES[Math.min(attempts, BACKOFF_MINUTES.length - 1)];
  }
}
```

```ts
import { Inject, Injectable, Logger } from '@nestjs/common';
import { applyDecision, Decision } from '../core/draft.entity';
import { REVIEW_CHANNEL_PORT, ReviewChannelPort } from '../core/ports/review-channel.port';
import { CuratorService } from '../curator/curator.service';
import { GeneratorService } from '../generator/generator.service';
import { SignalRepository } from '../persistence/signal.repository';
import { DraftRepository } from '../persistence/draft.repository';
import { CycleLogRepository, CycleOutcome } from '../persistence/cycle-log.repository';
import { TokenStore } from '../publisher/linkedin/token.store';

const LOOKBACK_DAYS = 7;

@Injectable()
export class CycleService {
  private readonly logger = new Logger(CycleService.name);

  constructor(
    private readonly curator: CuratorService,
    private readonly generator: GeneratorService,
    @Inject(REVIEW_CHANNEL_PORT) private readonly review: ReviewChannelPort,
    private readonly drafts: DraftRepository,
    private readonly signals: SignalRepository,
    private readonly cycles: CycleLogRepository,
    private readonly tokens: TokenStore,
  ) {}

  async runCycle(now: Date): Promise<CycleOutcome> {
    if (await this.tokens.needsReauth(now)) {
      await this.review.notify(
        'Hay que reconectar LinkedIn: abrí http://localhost:3000/auth/linkedin',
      );
    }

    const since = new Date(now.getTime() - LOOKBACK_DAYS * 24 * 60 * 60 * 1000);
    await this.curator.collect(since);

    // Score every unused signal, not only what this run captured: an idea sent on
    // Tuesday is still eligible on Friday.
    const pool = await this.curator.pendingSignals();
    const recentTopics = await this.curator.recentTopics(now);
    const best = this.curator.pickBest(pool, recentTopics, now);

    if (!best) {
      await this.review.notify(
        'Ciclo sin material: no hubo ningún hecho nuevo que valga un post. No publico relleno.',
      );
      await this.cycles.record(now, 'SKIPPED_NO_MATERIAL');
      return 'SKIPPED_NO_MATERIAL';
    }

    const draft = await this.generator.generateFor(best);
    await this.signals.markUsed(best.id);
    await this.review.sendForReview(draft);

    if (this.generator.lastViolations.length > 0) {
      await this.review.notify(
        `Ojo: el borrador quedó con violaciones de estilo: ${this.generator.lastViolations
          .map((violation) => `${violation.rule} (${violation.detail})`)
          .join(', ')}`,
      );
    }

    await this.cycles.record(now, 'RAN');
    return 'RAN';
  }

  async handleDecision(draftId: string, decision: Decision, now: Date): Promise<void> {
    const draft = await this.drafts.findById(draftId);
    if (!draft) {
      this.logger.warn(`Decision for unknown draft ${draftId}`);
      return;
    }

    const updated = applyDecision(draft, decision, now);
    await this.drafts.save(updated);

    if (decision.kind === 'revise') {
      const regenerated = await this.generator.regenerate(updated, decision.feedback);
      await this.review.sendForReview(regenerated);
    }
  }

  async expireStale(now: Date): Promise<number> {
    const stale = await this.drafts.findExpirable(now);
    for (const draft of stale) {
      await this.drafts.save({ ...draft, status: 'EXPIRED', updatedAt: now });
    }
    if (stale.length > 0) {
      await this.review.notify(`Vencieron ${stale.length} borrador(es) sin respuesta. No se publican.`);
    }
    return stale.length;
  }
}
```

```ts
const CYCLE_HOUR = 8;
const CYCLE_DAYS = [1, 3, 5]; // Monday, Wednesday, Friday

/** Most recent scheduled slot at or before `now`. */
export function lastScheduledSlot(now: Date): Date {
  const slot = new Date(now);
  slot.setHours(CYCLE_HOUR, 0, 0, 0);
  for (let back = 0; back < 8; back++) {
    const candidate = new Date(slot);
    candidate.setDate(slot.getDate() - back);
    if (CYCLE_DAYS.includes(candidate.getDay()) && candidate <= now) return candidate;
  }
  return slot;
}

@Injectable()
export class SchedulerService implements OnModuleInit {
  private readonly logger = new Logger(SchedulerService.name);

  constructor(
    private readonly cycle: CycleService,
    private readonly queue: PublishQueue,
    private readonly cycles: CycleLogRepository,
  ) {}

  async onModuleInit(): Promise<void> {
    await this.recoverMissedCycle(new Date());
  }

  @Cron('0 8 * * 1,3,5')
  async scheduledCycle(): Promise<void> {
    await this.cycle.runCycle(new Date());
  }

  @Cron(CronExpression.EVERY_5_MINUTES)
  async drainQueue(): Promise<void> {
    const now = new Date();
    await this.queue.drain(now);
    await this.cycle.expireStale(now);
  }

  async recoverMissedCycle(now: Date, slot: Date = lastScheduledSlot(now)): Promise<void> {
    const last = await this.cycles.lastCycleAt();
    if (last && last >= slot) {
      this.logger.log('Latest scheduled cycle already ran');
      return;
    }

    const ageMs = now.getTime() - slot.getTime();
    if (ageMs > GRACE_WINDOW_HOURS * HOUR_MS) {
      this.logger.log(`Missed cycle from ${slot.toISOString()} is stale, skipping`);
      await this.cycles.record(now, 'SKIPPED_STALE');
      return;
    }

    this.logger.log(`Recovering missed cycle from ${slot.toISOString()}`);
    await this.cycle.runCycle(now);
  }
}
```
